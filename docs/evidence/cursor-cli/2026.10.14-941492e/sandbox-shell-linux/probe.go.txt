func sandboxProbeInstrument(dir string) {
 shell := os.Getenv("SHELL")
 body, err := os.ReadFile(shell)
 if err != nil { panic(err) }
 logger := `import os,sys,json,time,ctypes,base64,socket
p=os.getppid()
env=dict(x.split('=',1) for x in open('/proc/%s/environ'%p,'rb').read().decode().split('\0') if x)
for k in list(env):
 if any(x in k.upper() for x in ['KEY','TOKEN','PASSWORD','SECRET']): env[k]='<REDACTED>'
record=dict(at_ns=time.time_ns(),pid=p,ppid=int(open('/proc/%s/stat'%p).read().split()[3]),argv=sys.argv[2:],cwd=os.getcwd(),env=env)
try:
 path='/proc/%s/fd/5'%p
 record['fd5_target']=os.readlink(path)
 if record['fd5_target'].startswith('socket:'):
  sock=socket.socket(fileno=os.dup(5))
  sock.settimeout(2)
  data=sock.recv(1048576,socket.MSG_PEEK)
  sock.close()
  fd=None
 else:
  fd=os.open(path,os.O_RDONLY|os.O_NONBLOCK)
 try:
  if fd is not None: data=os.pread(fd,1048576,0)
 except OSError:
  rd,wr=os.pipe()
  n=ctypes.CDLL(None,use_errno=True).tee(fd,wr,1048576,2)
  if n<0: raise OSError(ctypes.get_errno(),'tee')
  data=os.read(rd,n)
  os.close(rd);os.close(wr)
 if fd is not None: os.close(fd)
 record['fd5_base64']=base64.b64encode(data).decode()
 record['fd5_text']=data.decode()
except OSError as e: record['fd5_error']=str(e)
with open(sys.argv[1]+'/shell-%s.json'%p,'x') as f: json.dump(record,f,indent=2)
`
 script := filepath.Join(dir,"shell-record.py")
 if err := os.WriteFile(script, []byte(logger),0600); err != nil { panic(err) }
 prefix := "#!/bin/sh\n/usr/bin/python3 "+shellQuote(script)+" "+shellQuote(dir)+" \"$0\" \"$@\"\n"
 if err := os.WriteFile(shell, []byte(prefix+strings.TrimPrefix(string(body),"#!/bin/sh\n")),0700); err != nil { panic(err) }
}

func TestCursorSandboxProbe(t *testing.T) {
 r:=realCursorEnv(t)
 prompts:=[]string{
  "Shell invocation measurement. Use only the Shell tool, no edits, no other tools. First run exactly `ls`. Then run exactly `touch auto-allowed.txt`. Then run exactly `python3 -c 'print(424242)'`. Run all three as separate foreground tool calls. Do not combine or alter commands. Stop after these.",
  "Shell invocation measurement. Run exactly `sleep 4; printf background` using the Shell tool with block_until_ms: 0 so it continues in the background. Do not wait for it in the command. Then run exactly `printf foreground`. No other tools or commands.",
  "Shell invocation measurement. Issue two Shell tool calls together in parallel, not one combined shell command: `sleep 3; printf concurrent-A` and `sleep 3; printf concurrent-B`. Both block_until_ms: 0. No other commands.",
  "Shell invocation measurement. Issue TWO separate Shell tool calls together in parallel with IDENTICAL command text `sleep 3; printf identical`, both block_until_ms: 0. This deliberately tests duplicate invocations; do not deduplicate, combine, sequence or change the command text. No other commands.",
 }
 run:=r.run(t,realScenario{name:"sandbox-shell-linux-socket",mode:"auto",prompts:prompts,approve:func(card cursorTestCard)cursorTestDecision{return cursorTestDecision{Status:"ALLOWED"}}})
 report:=newRealReport(t)
 report.facts["approval_source"]="Controlled harness returns ALLOWED through the runner card path; not a human UI click."
 report.facts["kernel"]="Linux; real pinned CLI; temporary shell instrumentation only"
 r.save(t,"sandbox-shell-linux-socket","process",run,report)
 red:=newRealRedactor(r.key,run)
 files,_:=filepath.Glob(filepath.Join(run.raw,"shell-*.json"))
 var records []json.RawMessage
 for _,f:=range files { b,_:=os.ReadFile(f); records=append(records,json.RawMessage(red.text(string(b)))) }
 for _,name:=range []string{"stdin.jsonl","stdout.jsonl","stderr.txt","timeline.jsonl","exit.json"} {
 raw,err:=os.ReadFile(filepath.Join(run.raw,name)); if err==nil { os.WriteFile(filepath.Join(r.out,"sandbox-shell-linux-socket","process",name),[]byte(red.text(string(raw))),0644) }
 }
 b,_:=json.MarshalIndent(records,"","  ")
 if err:=os.WriteFile(filepath.Join(r.out,"sandbox-shell-linux-socket","shell.json"),b,0644);err!=nil{t.Fatal(err)}
 if len(files)==0{t.Fatal("no shell recordings")}
}
